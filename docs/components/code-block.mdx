## Import

```tsx
import { CodeBlockView } from '@lantos1618/better-ui/components';
```

## Basic Usage

```tsx
<CodeBlockView
  title="api/route.ts"
  language="typescript"
  code={`export async function POST(req: Request) {
  const { messages } = await req.json();
  return streamText({ model, messages });
}`}
/>
```

## Diff View

Pass a `diff` object with `before` and `after` strings. The block renders a
`before` / `diff` / `after` toggle, defaulting to the computed `diff` view.

```tsx
<CodeBlockView
  title="Changes"
  language="typescript"
  diff={{
    before: `const x = 1;`,
    after: `const x = 2;`,
  }}
/>
```

## Props

<ParamField path="code" type="string" required>
  The code content to display
</ParamField>

<ParamField path="language" type="string">
  Language label shown in the header (e.g. `'typescript'`, `'python'`, `'json'`)
</ParamField>

<ParamField path="title" type="string">
  Optional title (often the filename)
</ParamField>

<ParamField path="showLineNumbers" type="boolean" default="false">
  Show line numbers in the gutter
</ParamField>

<ParamField path="diff" type="{ before: string; after: string }">
  Enable diff view. Provide the previous and updated source; the component
  renders a `before` / `diff` / `after` toggle. In the diff view, removed lines
  are highlighted red and added lines green.
</ParamField>

<ParamField path="loading" type="boolean" default="false">
  Show a "Generating code…" placeholder instead of the code.
</ParamField>

<ParamField path="className" type="string">
  Additional CSS class applied to the root element.
</ParamField>

## Features

- **Copy button**: One-click copy with visual feedback
- **Syntax highlighting**: Language-aware highlighting for code blocks
- **Line numbers**: Optional line number gutter
- **Diff view**: `before` / `diff` / `after` toggle with green/red line highlighting
- **Loading state**: Built-in placeholder while code is streaming in
